MARCO INTRODUCTORIO

El mundo del hackeo ético y la ciberseguridad ha crecido exponencialmente, generando la necesidad de plataformas que permitan simular y competir en entornos controlados.

HACK BATTLE surge como una solución informática orientada a la gestión de competencias de hackeo simulado, permitiendo registrar jugadores, administrar herramientas, crear torneos, gestionar partidas por turnos y controlar rankings globales.

El sistema integra todos los conceptos aprendidos en la materia: tablas relacionales, triggers, vistas, procedimientos almacenados, subconsultas y seguridad de usuarios.

CÓDIGO SQL — TABLAS

-- Tabla Usuario
CREATE TABLE Usuario (
  id_usuario INT PRIMARY KEY IDENTITY(1,1),
  nombre_usuario VARCHAR(50) NOT NULL,
  correo VARCHAR(100) UNIQUE NOT NULL,
  contrasena VARCHAR(100) NOT NULL,
  fecha_registro DATE DEFAULT GETDATE()
);

-- Tabla Jugador
CREATE TABLE Jugador (
  id_jugador INT PRIMARY KEY IDENTITY(1,1),
  id_usuario INT NOT NULL,
  nivel INT DEFAULT 1,
  experiencia INT DEFAULT 0,
  victorias INT DEFAULT 0,
  derrotas INT DEFAULT 0,
  FOREIGN KEY (id_usuario)
    REFERENCES Usuario(id_usuario)
);
-- Tabla Herramienta
CREATE TABLE Herramienta (
  id_herramienta INT PRIMARY KEY IDENTITY(1,1),
  nombre VARCHAR(50) NOT NULL,
  tipo VARCHAR(20),
  daño INT,
  costo_energia INT,
  cooldown INT,
  nivel_requerido INT DEFAULT 1
);

-- Tabla Partida
CREATE TABLE Partida (
  id_partida INT PRIMARY KEY IDENTITY(1,1),
  id_jugador1 INT NOT NULL,
  id_jugador2 INT NOT NULL,
  ganador INT,
  turnos INT,
  fecha DATE DEFAULT GETDATE(),
  FOREIGN KEY (id_jugador1)
    REFERENCES Jugador(id_jugador)
);

TRIGGER — EJEMPLO DE CÓDIGO

trg_ActualizarRanking

CREATE TRIGGER trg_ActualizarRanking
ON Partida
AFTER INSERT
AS
BEGIN
  SET NOCOUNT ON;

  -- Actualizar victorias del ganador
  UPDATE Ranking
  SET victorias = victorias + 1
  FROM Ranking r
  INNER JOIN inserted i
    ON r.id_jugador = i.ganador;

  -- Actualizar derrotas del perdedor
  UPDATE Ranking
  SET derrotas = derrotas + 1
  FROM Ranking r
  INNER JOIN inserted i
    ON r.id_jugador =
    CASE WHEN i.ganador = i.id_jugador1
      THEN i.id_jugador2
      ELSE i.id_jugador1 END;
END;

trg_ValidarEnergia

CREATE TRIGGER trg_ValidarEnergia
ON MovimientoPartida
INSTEAD OF INSERT
AS
BEGIN
  SET NOCOUNT ON;

  IF EXISTS (
    SELECT 1 FROM inserted i
    INNER JOIN Herramienta h
      ON i.id_herramienta = h.id_herramienta
    INNER JOIN Jugador j
      ON i.id_jugador = j.id_jugador
    WHERE j.energia < h.costo_energia
  )
  BEGIN
    RAISERROR('Energía insuficiente',16,1);
    RETURN;
  END

  -- Si hay energía, insertar
  INSERT INTO MovimientoPartida
  SELECT * FROM inserted;
END;

SUBCONSULTAS Y SEGURIDAD

🔎 5 Subconsultas
  • Jugador con más victorias (MAX)
  • Herramienta más usada en partidas (COUNT + GROUP BY)
  • Jugadores con winrate superior al promedio (AVG)
  • Equipos que han ganado al menos un torneo (IN + EXISTS)
  • Partidas donde se usó la Bomba Lógica (subconsulta anidada)

-- Jugadores con winrate superior al promedio
SELECT nombre, victorias, derrotas
FROM Vista_RankingGlobal
WHERE winrate > (
  SELECT AVG(winrate)
  FROM Vista_RankingGlobal
);
